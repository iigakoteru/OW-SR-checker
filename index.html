// デバイスがスマホ・タブレット（iOS/Androidなど）かどうかを判定する便利関数 function isMobileDevice() { return /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent) || (navigator.maxTouchPoints && navigator.maxTouchPoints > 2); } // 単一チーム用スクショ function takeScreenshot(team) { const cardElement = document.getElementById(`card-team${team}`); html2canvas(cardElement, { backgroundColor: '#131b2e', scale: 2, useCORS: true }).then(canvas => { const imageUrl = canvas.toDataURL('image/png'); // スマホ・iPadの場合はポップアップ（モーダル）を表示して長押し保存させる if (isMobileDevice()) { showImageModal(imageUrl); } else { // PCの場合はこれまで通り自動ダウンロード const link = document.createElement('a'); link.download = `OW2_Team_${team}_SR_Result.png`; link.href = imageUrl; document.body.appendChild(link); link.click(); document.body.removeChild(link); } }).catch(err => { alert("画像の生成に失敗しました: " + err); }); } // VS対戦比較モード用 function takeMatchScreenshot() { const summaryElement = document.getElementById('vs-summary'); const wrapperElement = document.getElementById('teams-wrapper'); const container = document.createElement('div'); container.style.backgroundColor = '#0f141d'; container.style.padding = '24px'; container.style.width = '1200px'; const cloneSummary = summaryElement.cloneNode(true); const btnInClone = cloneSummary.querySelector('button'); if (btnInClone) btnInClone.remove(); const cloneWrapper = wrapperElement.cloneNode(true); container.appendChild(cloneSummary); const spacer = document.createElement('div'); spacer.style.height = '24px'; container.appendChild(spacer); container.appendChild(cloneWrapper); document.body.appendChild(container); html2canvas(container, { backgroundColor: '#0f141d', scale: 2, useCORS: true }).then(canvas => { document.body.removeChild(container); const imageUrl = canvas.toDataURL('image/png'); if (isMobileDevice()) { showImageModal(imageUrl); } else { const link = document.createElement('a'); link.download = `OW2_Scrim_Match_Analysis.png`; link.href = imageUrl; document.body.appendChild(link); link.click(); document.body.removeChild(link); } }).catch(err => { if (container.parentNode) document.body.removeChild(container); alert("画像の生成に失敗しました: " + err); }); }
